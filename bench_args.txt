# Аргументы бенчмарка быстродействия (bench.py / bench_regression.py).
# Формат: ключ = значение, # -- комментарий. Отдельно от RE_args.txt (атака).
#
# Бенчмарк ГОНЯЕТ бинарь на живой Raspberry Pi (throughput имеет смысл только на
# реальном железе), поэтому нужен настроенный доступ к Pi в .env (RE_PI_*).

# Папка с бинарём и его зависимостями (копируется на Pi в bench/samples).
sample-dir = ./samples

# Скрипт бенчмарка (копируется на Pi в bench/). Лежит в truth/ (эталон).
bench-script-path = ./truth/tools/bench.sh

# Сколько раз прогнать бенчмарк. Итог = МЕДИАНА (храним все прогоны + min/max/разброс).
# Один прогон шумит (планировщик, нагрев), поэтому N > 1.
reps = 5

# Обходить ли предохранитель bench.sh (governor/термалка/нагрузка). bench.sh без
# FORCE отказывается писать результат в плохих условиях. true -> передаём FORCE=1.
# ВНИМАНИЕ: на Pi без охлаждения замеры троттлят -> числа шумные, коридор регрессии
# шире. Входит в ключ сравнимости (forced и не-forced замеры несопоставимы).
force-benchmarking = true

# ship-сборка (bench.sh --dist). Задаётся ОТДЕЛЬНО: build-configuration=mixed не
# однозначно отображается в dist, поэтому не выводим автоматически. Входит в ключ
# сравнимости (dist и не-dist по кодогену считаются несопоставимыми).
dist = true
