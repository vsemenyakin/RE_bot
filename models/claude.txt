# Описание модели Claude. Ссылка на этот файл -- в models = ... (RE_args.txt).
#
# Два маршрута, симметрично: пара *_model + *_budget на каждый.
#   litellm_*      -- через OpenRouter (litellm_budget -- реальные деньги $)
#   subscription_* -- через подписку Claude Code, claude -p
#                     (subscription_budget -- лимит API-эквивалента -> --max-budget-usd,
#                      реальные деньги НЕ тратятся, расход из лимитов Pro)
# preferred_run_type выбирает активный маршрут. subscription требует свежий токен
# (claude auth login раз в ~28 дней) и только для ДОВЕРЕННОГО бинаря (контейнер
# Claude идёт с сетью и доступом к Pi).

name = claude

litellm_model  = openrouter/anthropic/claude-opus-4.8
litellm_budget = 3

# subscription_model -- имя для claude -p --model. Алиас "opus" = свежайший Opus.
# ~75%% окна Opus в Pro ≈ 10 (API-эквивалент), с запасом до session limit.
subscription_model  = opus
subscription_budget = 10

preferred_run_type = subscription
